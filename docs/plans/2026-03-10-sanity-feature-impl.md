# Sanity WorldFeature Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Implement a tabletop horror RPG faithful sanity/insanity WorldFeature with event-driven SAN loss tracking, temporary/indefinite insanity triggers, d10 bout of madness table, LLM-generated phobia/mania subjects, behavioral action restriction tags, and tick-based duration management.

**Architecture:** An exported `applySanityLoss()` utility function replaces all direct SAN mutations. It applies the delta, logs it, checks single-event and hourly cumulative triggers, rolls on the bout of madness table, and injects conditions. The WorldFeature `tick()` manages insanity durations and log cleanup. Phobia/mania subjects are generated by a small LLM call.

**Tech Stack:** TypeScript, Vitest, LangChain (generateText for small model calls)

// ===== State helpers =====

export function createEmptySanityState(): SanityCharacterState {
  return {
    sanLossLog: [],
    activeInsanity: null,
    persistentConditions: [],
  };
}

export function getSanityState(dgsm: DynamicGameStateManager, characterId: string): SanityCharacterState | undefined {
  return dgsm.getFeatureSceneState(FEATURE_ID, characterId) as SanityCharacterState | undefined;
}

export function setSanityState(dgsm: DynamicGameStateManager, characterId: string, state: SanityCharacterState): void {
  dgsm.setFeatureSceneState(FEATURE_ID, characterId, state);
}

export function recordSanLoss(state: SanityCharacterState, amount: number, gameTimeMinutes: number): void {
  state.sanLossLog.push({ amount, gameTimeMinutes });
}

export function getHourlyCumulativeLoss(state: SanityCharacterState, currentTimeMinutes: number): number {
  const windowStart = currentTimeMinutes - HOURLY_WINDOW_MINUTES;
  return state.sanLossLog
    .filter(entry => entry.gameTimeMinutes >= windowStart)
    .reduce((sum, entry) => sum + entry.amount, 0);
}

// ===== Dice helpers =====

function roll1dN(n: number): number {
  return Math.floor(Math.random() * n) + 1;
}

// ===== Trigger detection =====

export function checkTemporaryInsanityTrigger(sanLoss: number): boolean {
  return sanLoss >= SINGLE_LOSS_THRESHOLD;
}

export function checkIndefiniteInsanityTrigger(hourlyCumulativeLoss: number, currentSan: number): boolean {
  return hourlyCumulativeLoss >= Math.floor(currentSan / 5);
}

export function rollBoutOfMadness(): BoutTableEntry {
  const roll = roll1dN(10);
  return BOUT_OF_MADNESS_TABLE[roll - 1];
}

/** 1d10 hours → minutes */
export function rollTemporaryDuration(): number {
  return roll1dN(10) * 60;
}

/** 1d6 hours → minutes */
export function rollIndefiniteOnsetDelay(): number {
  return roll1dN(6) * 60;
}

/** 1d10 days → minutes */
export function rollIndefiniteDuration(): number {
  return roll1dN(10) * 1440;
}
```

// ===== Condition string builders =====

export function buildInsanityConditionString(
  insanityType: "temporary" | "indefinite" | "phobia" | "mania",
  boutType: BoutOfMadnessType,
  label: string,
  actionRestriction: ActionRestriction,
  subject?: string,
): string {
  const isPersistent = insanityType === "phobia" || insanityType === "mania";
  const subjectPart = subject ? ` (${subject})` : "";
  const tag = isPersistent ? "persistent" : `restriction:${actionRestriction}`;
  return `[Insanity:${insanityType}] ${label}${subjectPart} | ${tag}`;
}

// ===== Condition injection =====

export function injectInsanityCondition(
  dgsm: DynamicGameStateManager,
  characterId: string,
  conditionString: string,
  isPlayer: boolean,
): void {
  const state = dgsm.getState();
  let conditions: string[] | undefined;

  if (isPlayer) {
    conditions = state.playerCharacter?.status?.conditions;
  } else {
    const npc = state.npcCharacters?.find((n: any) => n.id === characterId);
    conditions = npc?.status?.conditions;
  }

  if (!conditions) return;
  conditions.push(conditionString);
}

export function removeInsanityCondition(
  dgsm: DynamicGameStateManager,
  characterId: string,
  isPlayer: boolean,
  keepPersistent = false,
): void {
  const state = dgsm.getState();
  let conditions: string[] | undefined;

  if (isPlayer) {
    conditions = state.playerCharacter?.status?.conditions;
  } else {
    const npc = state.npcCharacters?.find((n: any) => n.id === characterId);
    conditions = npc?.status?.conditions;
  }

  if (!conditions) return;

  const filtered = conditions.filter((c: string) => {
    if (!c.startsWith(INSANITY_CONDITION_PREFIX)) return true;
    if (keepPersistent && c.includes("| persistent")) return true;
    return false;
  });

  if (isPlayer) {
    (state.playerCharacter.status as any).conditions = filtered;
  } else {
    const npc = state.npcCharacters?.find((n: any) => n.id === characterId);
    if (npc?.status) {
      (npc.status as any).conditions = filtered;
    }
  }
}
```

/**
 * Unified SAN loss entry point. Replaces all direct SAN mutations.
 * Applies the delta, logs it, checks triggers, and injects conditions.
 *
 * @param delta Negative for loss, positive for recovery (recovery skips triggers)
 */
export function applySanityLoss(
  dgsm: DynamicGameStateManager,
  characterId: string,
  delta: number,
  isPlayer: boolean,
  gameDay: number,
  tickTime: string,
  source?: string,
): void {
  // Apply the actual SAN change
  if (isPlayer) {
    const player = dgsm.getState().playerCharacter;
    if (player?.status) {
      (player.status as any).sanity = Math.max(0, player.status.sanity + delta);
    }
  } else {
    dgsm.updateNpcSan(characterId, delta);
  }

  // Skip trigger logic for recovery (positive delta)
  if (delta >= 0) return;

  const lossAmount = Math.abs(delta);
  const currentTimeMinutes = computeGameTimeMinutes(gameDay, tickTime);

  // Get or create sanity state
  let sanState = getSanityState(dgsm, characterId);
  if (!sanState) {
    sanState = createEmptySanityState();
  }

  // Record loss
  recordSanLoss(sanState, lossAmount, currentTimeMinutes);

  // Skip insanity checks if already insane
  if (!sanState.activeInsanity) {
    // Get current SAN for indefinite check (SAN BEFORE this loss, so add back)
    const currentSan = isPlayer
      ? (dgsm.getState().playerCharacter?.status?.sanity ?? 0) + lossAmount
      : (dgsm.getNpcStats(characterId)?.san ?? 0) + lossAmount;

    const tempTrigger = checkTemporaryInsanityTrigger(lossAmount);
    const hourlyCumulative = getHourlyCumulativeLoss(sanState, currentTimeMinutes);
    const indefTrigger = checkIndefiniteInsanityTrigger(hourlyCumulative, currentSan);

    if (tempTrigger) {
      // Temporary insanity — immediate onset
      const bout = rollBoutOfMadness();
      const duration = rollTemporaryDuration();
      const desc = BOUT_DESCRIPTIONS[bout.boutType];

      sanState.activeInsanity = {
        insanityType: "temporary",
        boutType: bout.boutType,
        description: desc,
        actionRestriction: bout.actionRestriction,
        startMinutes: currentTimeMinutes,
        durationMinutes: duration,
        isActive: true,
      };

      // Inject condition
      const condStr = buildInsanityConditionString(
        "temporary", bout.boutType, bout.label, bout.actionRestriction,
      );
      injectInsanityCondition(dgsm, characterId, condStr, isPlayer);

      // Handle persistent phobia/mania
      if (bout.persistent) {
        sanState.persistentConditions.push({
          type: bout.boutType as "phobia" | "mania",
          subject: "unknown", // placeholder — LLM fills this in Task 6
          description: desc,
        });
        const persistentCondStr = buildInsanityConditionString(
          bout.boutType as "phobia" | "mania",
          bout.boutType, bout.label, bout.actionRestriction, "unknown",
        );
        injectInsanityCondition(dgsm, characterId, persistentCondStr, isPlayer);
      }
    } else if (indefTrigger) {
      // Indefinite insanity — delayed onset
      const bout = rollBoutOfMadness();
      const onsetDelay = rollIndefiniteOnsetDelay();
      const duration = rollIndefiniteDuration();
      const desc = BOUT_DESCRIPTIONS[bout.boutType];

      sanState.activeInsanity = {
        insanityType: "indefinite",
        boutType: bout.boutType,
        description: desc,
        actionRestriction: bout.actionRestriction,
        startMinutes: currentTimeMinutes,
        durationMinutes: duration,
        isActive: false,
        onsetDelayMinutes: onsetDelay,
      };

      // Persistent phobia/mania (stored but condition not injected until onset)
      if (bout.persistent) {
        sanState.persistentConditions.push({
          type: bout.boutType as "phobia" | "mania",
          subject: "unknown",
          description: desc,
        });
      }
    }
  }

  // Persist
  setSanityState(dgsm, characterId, sanState);
}

      ? `Based on this traumatic event, generate a single specific phobia (irrational fear) in 2-5 words. Event: "${triggerContext}" at location "${sceneId}". Reply with ONLY the phobia subject, nothing else. Example: "darkness", "tentacles", "confined spaces", "drowning".`

// ===== Tracked characters helper (reuse from stamina pattern) =====

function getTrackedCharacters(dgsm: DynamicGameStateManager): Array<{
  characterId: string;
  isPlayer: boolean;
}> {
  const state = dgsm.getState();
  const result: Array<{ characterId: string; isPlayer: boolean }> = [];

  if (state.playerCharacter?.id) {
    result.push({ characterId: state.playerCharacter.id, isPlayer: true });
  }

  for (const npcId of Object.keys(state.npcLocations)) {
    result.push({ characterId: npcId, isPlayer: false });
  }

  return result;
}

// ===== Exported feature =====

export const sanityFeature: WorldFeature = {
  id: FEATURE_ID,
  description: "tabletop horror RPG sanity system — temporary/indefinite insanity, bout of madness, phobias/manias",

  planningPrompt: `## Sanity / Insanity
- Characters who lose 5+ SAN from a single event may suffer **temporary insanity** (lasts hours).
- Characters who lose ≥1/5 of their current SAN within one game hour may suffer **indefinite insanity** (lasts days, delayed onset).
- Insanity manifests as a Bout of Madness: amnesia, violence, paranoia, fainting, fleeing, hysterics, phobia, or mania.
- Active insanity imposes behavioral restrictions shown in character conditions (e.g. restriction:flee_only, restriction:incapacitated).
- Persistent phobias and manias remain after the insanity episode resolves.
- You do NOT control insanity directly — it is tracked automatically by the sanity system.
- When planning for a character with active insanity, respect their action restriction.`,

  stateDescription(dgsm: DynamicGameStateManager): string {
    const allStates = dgsm.getFeatureState(FEATURE_ID) as Record<string, SanityCharacterState>;
    const lines: string[] = [];

    for (const [characterId, sanState] of Object.entries(allStates)) {
      if (!sanState) continue;

      if (sanState.activeInsanity?.isActive) {
        const ins = sanState.activeInsanity;
        lines.push(`- ${characterId}: ${ins.insanityType} insanity (${ins.boutType}) — ${ins.description} | restriction:${ins.actionRestriction}`);
      }

      for (const pc of sanState.persistentConditions) {
        lines.push(`- ${characterId}: ${pc.type} — ${pc.subject}`);
      }
    }

    return lines.length > 0 ? "Mental state:\n" + lines.join("\n") : "";
  },

  tick(dgsm: DynamicGameStateManager, runtime: TickRuntimeContext): void {
    const characters = getTrackedCharacters(dgsm);
    const currentTimeMinutes = computeGameTimeMinutes(runtime.gameDay, runtime.tickTime);

    for (const { characterId, isPlayer } of characters) {
      const sanState = getSanityState(dgsm, characterId);
      if (!sanState) continue;

      let changed = false;

      // 1. Check indefinite insanity onset
      if (sanState.activeInsanity && !sanState.activeInsanity.isActive) {
        const onsetTime = sanState.activeInsanity.startMinutes + (sanState.activeInsanity.onsetDelayMinutes ?? 0);
        if (currentTimeMinutes >= onsetTime) {
          sanState.activeInsanity.isActive = true;
          // Inject condition now
          const bout = BOUT_OF_MADNESS_TABLE.find(b => b.boutType === sanState.activeInsanity!.boutType);
          if (bout) {
            const condStr = buildInsanityConditionString(
              "indefinite", bout.boutType, bout.label, bout.actionRestriction,
            );
            injectInsanityCondition(dgsm, characterId, condStr, isPlayer);
          }
          changed = true;
        }
      }

      // 2. Check insanity expiry
      if (sanState.activeInsanity?.isActive) {
        const startTime = sanState.activeInsanity.insanityType === "indefinite"
          ? sanState.activeInsanity.startMinutes + (sanState.activeInsanity.onsetDelayMinutes ?? 0)
          : sanState.activeInsanity.startMinutes;
        const expiryTime = startTime + sanState.activeInsanity.durationMinutes;

        if (currentTimeMinutes >= expiryTime) {
          // Remove active insanity condition but keep persistent ones
          removeInsanityCondition(dgsm, characterId, isPlayer, true);
          sanState.activeInsanity = null;
          changed = true;
        }
      }

      // 3. Cleanup old sanLossLog
      const windowStart = currentTimeMinutes - HOURLY_WINDOW_MINUTES;
      const beforeLen = sanState.sanLossLog.length;
      sanState.sanLossLog = sanState.sanLossLog.filter(e => e.gameTimeMinutes >= windowStart);
      if (sanState.sanLossLog.length !== beforeLen) changed = true;

      if (changed) {
        setSanityState(dgsm, characterId, sanState);
      }
    }
  },
};
